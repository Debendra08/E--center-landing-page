// Paste this into Firebase Console → Firestore Database → Rules → Publish
// Public visitors can READ products and CREATE leads (submit interest forms),
// but only a logged-in admin (you) can write/edit/delete products or read leads.

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    match /products/{productId} {
      allow read: if true;
      allow write: if request.auth != null;
    }

    match /leads/{leadId} {
      allow create: if true;
      allow read, update, delete: if request.auth != null;
    }

  }
}
